WEBSOCKET /ws (REALTIME INTEGRATION)

Quick Start (JS)
const ws = new WebSocket("ws://<host>:9019/ws");
ws.onopen = () => ws.send(JSON.stringify({
  message: "Hello",
  phone_number: "9999999999",
  call_uuid: "abc-123"
}));
ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  if (msg.type === "chat") console.log(msg.reply);
};
Payload (same as /chat)
{
  "message": "Hello",
  "phone_number": "9999999999",
  "call_uuid": "abc-123",

Python (websockets)
import asyncio
import json
import websockets

async def main():
    async with websockets.connect("ws://<host>:9019/ws") as ws:
        await ws.send(json.dumps({
            "message": "Hello",
            "phone_number": "9999999999",
            "call_uuid": "abc-123"
        }))
        data = json.loads(await ws.recv())
        print(data)

asyncio.run(main())

WEBSOCKET INTEGRATION (UI + REST FALLBACK)

Frontend (WS + REST fallback)
async function sendChat(payload) {
  if (!window.WebSocket) return fetchChat(payload);
  return new Promise((resolve, reject) => {
    const ws = new WebSocket("ws://<host>:9019/ws");
    ws.onopen = () => ws.send(JSON.stringify(payload));
    ws.onmessage = (evt) => {
      const msg = JSON.parse(evt.data);
      if (msg.type === "chat") resolve(msg);
      else reject(msg);
      ws.close();
    };
    ws.onerror = () => fetchChat(payload).then(resolve).catch(reject);
  });
}

async function fetchChat(payload) {
  const res = await fetch("http://<host>:9019/chat", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload)
  });
  return res.json();
}
Notes
  • Use wss:// when running behind HTTPS.
  • Keep the same call_uuid for a session.
  • The UI uses WebSocket by default and falls back to REST.
